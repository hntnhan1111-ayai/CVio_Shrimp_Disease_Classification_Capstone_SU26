#
# The pinned versions below are the known-good combination that successfully

#
# Usage:
#   python -m venv .venv-export
#   .venv-export\Scripts\activate            # Windows
#   source .venv-export/bin/activate          # Linux/macOS
#   python -m pip install -r requirements-export-litert.txt
#
# Then run:
#   python scripts/06_export_litert_fp32_fp16.py --weights <best.pt> --out-dir export --imgsz 224
# ---------------------------------------------------------------------------
